🔌 1. Montagem Física das Setas (Papel Alumínio $\rightarrow$ ESP32)

Cada uma das 4 setas é um botão de chão (placas de papelão revestidas com papel alumínio, separadas por tirinhas de EVA nas bordas):

  • Placa de Baixo: Ligada ao GND comum (Terra compartilhado por todas as setas).
  • Placa de Cima: Ligada ao pino de sinal GPIO correspondente no ESP32.
  • Não precisa de resistor externo: O ESP32 utiliza resistor interno de INPUT_PULLUP.
[ ESP32 ] ┌──────────────────────────────────────────────────────────────┐ │ D15 (GPIO 15) ────► 🟠 Fio Laranja ──► Seta CIMA (UP) │ │ D18 (GPIO 18) ────► 🟢 Fio Verde ──► Seta BAIXO (DW) │ │ D4 (GPIO 4) ────► ⚪🟠 Branco-Laranja ──► Seta DIREITA(RT)│ │ D5 (GPIO 5) ────► 🔵 Fio Azul ──► Seta ESQUERDA(LT│ │ GND ────► 🟤 Fio Marrom Comum ──► Placas de Baixo │ └──────────────────────────────────────────────────────────────┘
Direção Pino ESP32 Cor do Fio (Cabo de Rede) Ação ao Pisar
⬆️ CIMA (UP) D15 🟠 Laranja Encosta no GND e dispara interrupção em D15
⬇️ BAIXO (DOWN) D18 🟢 Verde Encosta no GND e dispara interrupção em D18
➡️ DIREITA (RIGHT) D4 ⚪🟠 Branco-Laranja Encosta no GND e dispara interrupção em D4
⬅️ ESQUERDA (LEFT) D5 🔵 Azul Encosta no GND e dispara interrupção em D5
⏚ TERRA COMUM GND 🟤 Marrom Interliga a placa inferior de todas as 4 setas

💻 2. Código Oficial do ESP32 (FreeRTOS + MQTT)

Grave este código no ESP32 usando o Arduino IDE com a biblioteca PubSubClient instalada.

O que você precisa alterar no código abaixo:
1. WIFI_SSID e WIFI_PASSWORD com o nome e senha do seu Wi-Fi/Roteador.
2. MQTT_BROKER com o IP do computador que está rodando o Mosquitto (ex: "192.168.3.55"), ou use "broker.hivemq.com" para testes via internet. 
#include <WiFi.h>
#include <PubSubClient.h>

// =================================================================
// 1. CONFIGURAÇÕES DE REDE E BROKER (ALTERE AQUI!)
// =================================================================
const char* WIFI_SSID     = "NOME_DO_SEU_WIFI";
const char* WIFI_PASSWORD = "SENHA_DO_SEU_WIFI";

// IP do notebook rodando o Mosquitto (ou "broker.hivemq.com")
const char* MQTT_BROKER   = "192.168.3.55"; 
const int   MQTT_PORT     = 1883;

// Tópicos oficiais do jogo
const char* TOPIC_PASSOS  = "str/ddr/artur/passos";
const char* TOPIC_LEGACY  = "str/ddr/passos";

// Pinagem Oficial:
#define PIN_UP     15  // D15 -> Cima
#define PIN_DOWN   18  // D18 -> Baixo
#define PIN_RIGHT   4  // D4  -> Direita
#define PIN_LEFT    5  // D5  -> Esquerda

// =================================================================
// 2. FREERTOS E FILAS (REGRAS DE SOFTWARE EM TEMPO REAL)
// =================================================================
enum DirecaoSeta { SETA_UP = 0, SETA_DOWN = 1, SETA_LEFT = 2, SETA_RIGHT = 3 };

typedef struct {
  DirecaoSeta direcao;
  uint32_t    timestamp;
} EventoPasso_t;

QueueHandle_t filaPassos = NULL;
volatile uint32_t ultimoClique[4] = {0, 0, 0, 0};
const char* NOMES_SETAS[] = { "UP", "DOWN", "LEFT", "RIGHT" };

// Debounce por software na ISR (120ms para eliminar trepidação do papel alumínio)
void IRAM_ATTR isrGenerica(DirecaoSeta dir) {
  uint32_t agora = xTaskGetTickCountFromISR() * portTICK_PERIOD_MS;
  if (agora - ultimoClique[dir] > 120) {
    ultimoClique[dir] = agora;
    EventoPasso_t ev = { dir, agora };
    BaseType_t xHigherPriorityTaskWoken = pdFALSE;
    xQueueSendFromISR(filaPassos, &ev, &xHigherPriorityTaskWoken);
    if (xHigherPriorityTaskWoken) {
      portYIELD_FROM_ISR();
    }
  }
}

void IRAM_ATTR isrUp()    { isrGenerica(SETA_UP); }
void IRAM_ATTR isrDown()  { isrGenerica(SETA_DOWN); }
void IRAM_ATTR isrLeft()  { isrGenerica(SETA_LEFT); }
void IRAM_ATTR isrRight() { isrGenerica(SETA_RIGHT); }

// =================================================================
// 3. TAREFA MQTT NO FREERTOS (FUGA DO POLLING)
// =================================================================
WiFiClient espClient;
PubSubClient mqttClient(espClient);

void reconectarMQTT() {
  while (!mqttClient.connected()) {
    String clientId = "ESP32-DancePad-" + String(random(0xffff), HEX);
    Serial.print("Conectando ao Broker MQTT...");
    if (mqttClient.connect(clientId.c_str())) {
      Serial.println(" CONECTADO!");
    } else {
      Serial.print(" Falhou, rc=");
      Serial.print(mqttClient.state());
      Serial.println(". Tentando novamente em 1.5s...");
      vTaskDelay(pdMS_TO_TICKS(1500));
    }
  }
}

void TaskMQTT(void* pvParameters) {
  EventoPasso_t evento;
  char payload[64];

  for (;;) {
    // Bloqueia consumindo 0% de CPU até o jogador pisar na seta
    if (xQueueReceive(filaPassos, &evento, portMAX_DELAY) == pdTRUE) {
      if (!mqttClient.connected()) {
        reconectarMQTT();
      }
      mqttClient.loop();

      snprintf(payload, sizeof(payload), "{\"seta\":\"%s\",\"ts\":%lu}",
               NOMES_SETAS[evento.direcao], evento.timestamp);

      Serial.print("Pisada no tapete! Enviando: ");
      Serial.println(payload);

      mqttClient.publish(TOPIC_PASSOS, payload);
      mqttClient.publish(TOPIC_LEGACY, payload);
    }
  }
}

// =================================================================
// 4. SETUP E INICIALIZAÇÃO
// =================================================================
void setup() {
  Serial.begin(115200);
  delay(500);
  Serial.println("\n=== Evil Dance ESP32 Iniciando ===");

  pinMode(PIN_UP, INPUT_PULLUP);
  pinMode(PIN_DOWN, INPUT_PULLUP);
  pinMode(PIN_LEFT, INPUT_PULLUP);
  pinMode(PIN_RIGHT, INPUT_PULLUP);

  attachInterrupt(digitalPinToInterrupt(PIN_UP), isrUp, FALLING);
  attachInterrupt(digitalPinToInterrupt(PIN_DOWN), isrDown, FALLING);
  attachInterrupt(digitalPinToInterrupt(PIN_LEFT), isrLeft, FALLING);
  attachInterrupt(digitalPinToInterrupt(PIN_RIGHT), isrRight, FALLING);

  filaPassos = xQueueCreate(16, sizeof(EventoPasso_t));

  Serial.print("Conectando ao Wi-Fi: ");
  Serial.println(WIFI_SSID);
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(300);
    Serial.print(".");
  }
  Serial.println("\nWi-Fi Conectado! IP do ESP32: " + WiFi.localIP().toString());

  mqttClient.setServer(MQTT_BROKER, MQTT_PORT);

  xTaskCreatePinnedToCore(
    TaskMQTT,
    "TaskMQTT",
    4096,
    NULL,
    2,
    NULL,
    1
  );

  Serial.println("Pronto! Pise nas setas...");
}

void loop() {
  vTaskDelay(pdMS_TO_TICKS(1000));
}

📡 3. Configuração do Broker Mosquitto Local

O Mosquitto deve abrir duas portas: 1883 (TCP para o ESP32) e 9001 (WebSockets para a interface Web).

1. Arquivo str_mosquitto.conf

allow_anonymous true

# Porta para o ESP32 conectar via Wi-Fi (MQTT TCP padrão)
listener 1883 0.0.0.0
protocol mqtt

# Porta para o Site Web conectar via Navegador (WebSockets)
listener 9001 0.0.0.0
protocol websockets

2. Como Iniciar no Terminal (PowerShell no Windows)

  1. Se o serviço do Windows estiver ativo, pare-o no Gerenciador de Tarefas (Ctrl+Shift+Esc) $\rightarrow$ aba Serviços $\rightarrow$ botão direito em mosquitto $\rightarrow$ Parar.
  2. No terminal, execute o comando com o operador &: 
    & "C:\Program Files\mosquitto\mosquitto.exe" -c str_mosquitto.conf -v
  3. O terminal exibirá: 
    Opening ipv4 listen socket on port 1883.
Opening ipv4 listen socket on port 9001.
mosquitto version 2.1.2 running

🕹️ 4. Conectando a Interface Web (Evil Dance)

  1. Inicie o jogo no navegador acessando http://localhost:4173/ (ou pelo servidor local via npm start).
  2. No canto superior direito da tela, clique no botão MQTT.
  3. Clique no botão de atalho 💻 Mosquitto Local (WS):
    • URL: ws://localhost:9001 (ou ws://IP_DO_NOTE:9001 se outro PC for usar a tela).
    • Tópico: str/ddr/artur/passos (ou str/ddr/passos).
  4. Clique em Conectar. O indicador no topo mudará para verde: Tapete MQTT!
  5. Teste de Hardware sem iniciar a música: Se alguém pisar no tapete agora, a tela exibe imediatamente um aviso: 👟 Pisada recebida do broker: UP (Conexão OK!).

🎯 5. Guia de Calibração de Latência (Offset / Freshness)

O tapete de papelão, a compressão mecânica do EVA e o Wi-Fi têm um tempo de resposta de 20 ms a 70 ms. Para que as setas registrem PERFECT exatamente quando o pé encosta:

Como Calibrar no Jogo:
1. Na tela de seleção de músicas, abra o menu sanfona "Ajustar sincronia e ritmo" (abaixo do botão verde Vamos Jogar).
2. Localize o controle deslizante Compensação (Offset em ms):
  • Se a pessoa pisa no tempo certo, mas o jogo diz "ATRASOU": Aumente o valor para positivo (ex: +40 ms a +80 ms).
  • Se o jogo diz "ADIANTOU": Reduza para negativo (ex: -30 ms).
3. O valor fica gravado automaticamente no navegador para todas as partidas seguintes!

🧠 6. Colinha Teórica de STR (Perguntas do Prof. Evilasio)

1. Por que não usar polling no ESP32?
Porque polling consome 100% de CPU verificando pinos. Usamos Interrupções de Hardware (ISRs) tratando as pisadas como tarefas esporádicas. A tarefa consumidora no FreeRTOS fica bloqueada em xQueueReceive (0% CPU) até que ocorra um evento físico.
2. Como é garantida a integridade entre tarefas sem variáveis globais?
Seguindo a regra de ouro das Aulas 11 a 13 do professor, usamos Filas do FreeRTOS (Queues) via xQueueSendFromISR e xQueueReceive. Não há variáveis globais compartilhadas desprotegidas e nenhum Mutex é chamado dentro da ISR.
3. Qual o papel do Broker MQTT no projeto?
Ele implementa o padrão Publish/Subscribe garantindo desacoplamento espacial (a interface Web não sabe o que é um ESP32 nem onde ele está) e desacoplamento temporal.
4. O que é o Offset à luz da teoria de STR?
Trata-se da garantia da Atualidade dos Dados (Freshness) e avaliação de Deadlines Relativos (Firm Real-Time): se o dado da pisada chega fora da janela de julgamento, seu valor de pontuação cai a zero. O slider de offset compensa o atraso mecânico e de transporte.
